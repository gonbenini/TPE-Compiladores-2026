program program_2
// en este test vemos como el Lexer reconoce Tokens en situaciones de:
// - iterador from.
// - asignaciones en expresiones.
// - declaracion de orden de parametros en llamados a funciones.
// - conversiones de punto flotante a entero.



//declaracion constante
longint a, b;
singlef c;

//declaracion funcion
longint function f1(singlef p1) begin

    ret p1+10$l;
end;

longint function f2(longint p1, longint p2) begin

    ret p1 + p2;
end;


//sentencias ejecutables
begin
    //iteradores
    from a := 0$l to 9$l by 1$l repeat
        pout("Hola mundo");
    
    from b := 0$l to 9$l by 3$l repeat
    begin
        pout("Hola mundo");
    end;

    //asignacion en expresiones
    a := b = 2$l;
    b := f1(c=19.3s+3);
    c := a + b=4$l;

    if(a=b==3$l)
        pout(b);
    
    if(a=b+c>=3$l)
        pout(b);

    //orden de parametros
    pout(    f2(a, b)   [2$l, 1$l]   ); 

    b = tol(c / 3.4s-4);
end
