program test1_basico
typedef estado = [1$l, 2$l, 3$l];
longint a, b, c, i, r;
singlef f1, f2, fres;
estado est;

begin
    // Aritmetica entera y flotante (incluye negativos)
    a := 10$l;
    b := -2$l;
    c = a * 2$l + b / 2$l - 4$l;
    pout("Resultado entero c:");
    pout(c);

    f1 := 6.0;
    f2 := 2.5;
    fres = f1 + f2 * 2.0 - 1.0;
    pout("Resultado flotante fres:");
    pout(fres);

    // Conversion explicita valida tol() sin perdida de informacion
    r := tol(f1);
    pout("Conversion tol(6.0):");
    pout(r);

    // Enumeracion (Tema 23): asignacion valida por constante y por expresion
    est = 1$l;
    pout("Enum inicial:");
    pout(est);
    est := est + 1$l;
    pout("Enum actualizado:");
    pout(est);

    // Asignacion en linea dentro de expresion y condicion (Temas 17-18)
    r := a = 5$l + 3$l;
    pout("Asignacion en linea (a y r):");
    pout(a);
    pout(r);

    // Seleccion if / else y comparadores con anidamiento de iteracion
    if (a = 4$l == 4$l)
    begin
        pout("Entro al IF con asignacion en condicion");
        from i := 1$l to 3$l by 1$l repeat
        begin
            if (i != 2$l)
            begin
                pout("Iteracion distinta de 2:");
                pout(i);
            end;
            else
            begin
                pout("Iteracion igual a 2");
            end;
            end_if;
        end;
    end;
    else
    begin
        pout("No deberia entrar al ELSE");
    end;
    end_if;

    if (f1 >= f2)
        pout("Comparacion flotante ok");
    end_if;
end;
