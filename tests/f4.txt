program test_from_bordes
longint i, suma, cant;

begin
    // 1) Ascendente paso 1: 1, 2, 3, 4 -> suma=10, cant=4
    suma := 0$l;
    cant := 0$l;
    from i := 1$l to 4$l by 1$l repeat
    begin
        suma := suma + i;
        cant := cant + 1$l;
    end;
    pout("1) Asc paso 1 (esp suma=10, cant=4):");
    pout(suma);
    pout(cant);

    // 2) Ascendente con overshoot (no cae exacto en limite): 1, 3, 5 -> suma=9, cant=3
    suma := 0$l;
    cant := 0$l;
    from i := 1$l to 6$l by 2$l repeat
    begin
        suma := suma + i;
        cant := cant + 1$l;
    end;
    pout("2) Asc overshoot (esp suma=9, cant=3):");
    pout(suma);
    pout(cant);

    // 3) Una sola iteracion: inicio == fin (5) y paso mayor al rango (1)
    suma := 0$l;
    cant := 0$l;
    from i := 5$l to 5$l by 1$l repeat
    begin
        suma := suma + i;
        cant := cant + 1$l;
    end;
    from i := 1$l to 3$l by 10$l repeat
        cant := cant + 1$l;
    pout("3) Borde 1 iter (esp suma=5, cant=2):");
    pout(suma);
    pout(cant);

    // 4) Rango negativo y cruce por cero: -6,-4,-2 (suma=-12) y -2,-1,0,1,2 (cant=5, suma=0)
    suma := 0$l;
    from i := -6$l to -2$l by 2$l repeat
        suma := suma + i;
    pout("4a) Neg asc (esp suma=-12):");
    pout(suma);

    suma := 0$l;
    cant := 0$l;
    from i := -2$l to 2$l by 1$l repeat
    begin
        suma := suma + i;
        cant := cant + 1$l;
    end;
    pout("4b) Cruce cero (esp suma=0, cant=5):");
    pout(suma);
    pout(cant);

    // 5) Descendente paso -1: 4, 3, 2, 1 -> suma=10, cant=4
    suma := 0$l;
    cant := 0$l;
    from i := 4$l to 1$l by -1$l repeat
    begin
        suma := suma + i;
        cant := cant + 1$l;
    end;
    pout("5) Desc paso -1 (esp suma=10, cant=4):");
    pout(suma);
    pout(cant);

    // 6) Descendente con overshoot y cruce por cero: 3, 1, -1 -> suma=3, cant=3
    suma := 0$l;
    cant := 0$l;
    from i := 3$l to -2$l by -2$l repeat
    begin
        suma := suma + i;
        cant := cant + 1$l;
    end;
    pout("6) Desc overshoot (esp suma=3, cant=3):");
    pout(suma);
    pout(cant);
end;
